% Glyphes absents de Latin Modern (lettres grecques, indices, flèches) :
% repli sur DejaVu, plutôt qu'un caractère qui disparaît en silence.
\directlua{luaotfload.add_fallback("repli", {"DejaVu Serif:mode=harf;", "DejaVu Sans:mode=harf;", "Latin Modern Math:mode=harf;"})}
\setmainfont{Latin Modern Roman}[RawFeature={fallback=repli}]
\setsansfont{Latin Modern Sans}[RawFeature={fallback=repli}]
\directlua{luaotfload.add_fallback("replimono", {"DejaVu Sans:mode=harf;", "Latin Modern Math:mode=harf;"})}
\setmonofont{DejaVu Sans Mono}[Scale=0.86,RawFeature={fallback=replimono}]
